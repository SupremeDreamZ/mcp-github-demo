function makeDataTexture(kind='carbon',size=256){
 const data=new Uint8Array(size*size*4);
 for(let y=0;y<size;y++)for(let x=0;x<size;x++){
  const i=(y*size+x)*4; let r=0,g=0,b=0,a=255;
  if(kind==='carbon'){
   const u=x/size*48,v=y/size*48;
   const a1=Math.pow(.5+.5*Math.sin((u+v)*Math.PI),8), a2=Math.pow(.5+.5*Math.sin((u-v)*Math.PI),8);

  } else if(kind==='carbonRough'){
   const u=x/size*48,v=y/size*48; const weave=.5+.5*Math.sin((u+v)*Math.PI)*Math.sin((u-v)*Math.PI); r=g=b=Math.floor(80+55*weave);
  } else if(kind==='carbonNormal'){
   const u=x/size*48,v=y/size*48; const nx=Math.sin((u+v)*Math.PI)*.34+Math.sin((u-v)*Math.PI)*.34; const ny=Math.cos((u+v)*Math.PI)*.34-Math.cos((u-v)*Math.PI)*.34; r=Math.floor(128+nx*90);g=Math.floor(128+ny*90);b=245;
  } else if(kind==='brushed'){

   r=g=b=Math.floor(100+n*18+scratch);
  } else if(kind==='glassRough'){
   const dx=x-size*.56,dy=y-size*.48,rr=Math.hypot(dx,dy),ang=Math.atan2(dy,dx);
   const ridge=Math.max(0,Math.sin(rr*.43+ang*2.2)-.9)*90;
   const dust=Math.abs(Math.sin(x*17.17+y*41.31)*43758.5453)%1>.986?72:0;

  }
